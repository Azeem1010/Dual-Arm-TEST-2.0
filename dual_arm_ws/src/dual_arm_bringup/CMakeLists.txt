cmake_minimum_required(VERSION 3.16)
project(dual_arm_bringup)
find_package(ament_cmake REQUIRED)
find_package(moveit_ros_move_group REQUIRED)
add_executable(move_group_lifetime src/move_group_lifetime.cpp)
target_compile_features(move_group_lifetime PRIVATE cxx_std_17)
ament_target_dependencies(move_group_lifetime moveit_ros_move_group)
target_link_libraries(move_group_lifetime moveit_ros_move_group::moveit_move_group_capabilities_base ${CMAKE_DL_LIBS})
install(TARGETS move_group_lifetime DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
