cmake_minimum_required(VERSION 3.16)
project(dual_arm_gripper)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(dual_arm_suction SHARED src/suction_system.cpp)
target_compile_features(dual_arm_suction PRIVATE cxx_std_20)
target_compile_options(dual_arm_suction PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(dual_arm_suction PRIVATE include)
target_link_libraries(dual_arm_suction PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS dual_arm_suction LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(gripper_policy_test test/policy_test.cpp)
  target_compile_features(gripper_policy_test PRIVATE cxx_std_20)
  target_include_directories(gripper_policy_test PRIVATE include)
  add_test(NAME gripper_policy COMMAND gripper_policy_test)
endif()
ament_package()
